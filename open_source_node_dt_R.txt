library(rpart)

# Printing the R formula behind the decision tree
print("Formula or the R decision tree:")
print(dm_model_formula)
# Dimensions of the training data set
cat("Dimensions of dm_traindf:",dim(dm_traindf),"\n")
cat("Column names of dm_traindf:",colnames(dm_traindf),"\n")
# Fitting the decision to the training data frame
dm_model <- rpart(dm_model_formula,data=dm_traindf, method = "class")
# View model output
summary(dm_model)
