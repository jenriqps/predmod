from sklearn.neural_network import MLPClassifier

# Training
print("List of categorical and interval input variables:")
print(dm_input)
X_train = dm_traindf.loc[:, dm_input]
print("Features of X_train:")
print(X_train.shape)
print(X_train.columns)

# Dummy encode class variables
X_train_enc = pd.get_dummies(X_train, columns=dm_class_input, drop_first=True)
print("Features of X_train_enc:")
print(X_train_enc.shape)
print(X_train_enc.columns)

# Create y (labels)
y_train = dm_traindf[dm_dec_target]
print("Features of y_train:")
print(y_train.shape)

# Fitting the MLP neural network to the training data frame
dm_model = MLPClassifier(random_state = 2025,max_iter=1000,alpha = 0.05,activation='logistic',hidden_layer_sizes=[40])
dm_model.fit(X_train_enc, y_train)

# Score full data: posterior probabilities
X_test = dm_inputdf.loc[:, dm_input]
X_test_enc = pd.get_dummies(X_test, columns=dm_class_input, drop_first=True)
print("Features of X_test_enc:")
print(X_test_enc.shape)
print(X_test_enc.columns)
dm_scoreddf_prob = pd.DataFrame(dm_model.predict_proba(X_test_enc),columns=dm_predictionvar)
print("Features of dm_scoreddf_prob:")
print(dm_scoreddf_prob.shape)
print(dm_scoreddf_prob.columns)

# Score full data: class prediction
# All the data frame is scored. SAS wil identify internally which record is from the training and validate data frame.
dm_scoreddf_class = pd.DataFrame(dm_model.predict(X_test_enc),columns=[dm_classtarget_intovar])
print("Features of dm_scoreddf_class:")
print(dm_scoreddf_class.shape)
print(dm_scoreddf_class.columns)

# Column merge posterior probabilities and class prediction
print("debug")
dm_scoreddf = pd.concat([dm_scoreddf_prob, dm_scoreddf_class], axis=1)
print("Features of dm_scoreddf:")
print(dm_scoreddf.shape)
print(dm_scoreddf.columns)
